from datetime import datetime, timedelta, timezone
from jose import jwt, JWTError
from passlib.context import CryptContext
from config import JWT_SECRET, JWT_ALGORITHM, JWT_EXPIRE_HOURS

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")

def hash_password(password: str) -> str:
    return pwd_context.hash(password)

def verify_password(plain: str, hashed: str) -> bool:
    return pwd_context.verify(plain, hashed)

def create_token(user_id: int) -> str:
    expire = datetime.now(timezone.utc) + timedelta(hours=JWT_EXPIRE_HOURS)
    return jwt.encode(
        {"sub": str(user_id), "exp": expire},
        JWT_SECRET,
        algorithm=JWT_ALGORITHM,
    )

def decode_token(token: str) -> int | None:
    try:
        payload = jwt.decode(token, JWT_SECRET, algorithms=[JWT_ALGORITHM])
        return int(payload["sub"])
    except (JWTError, KeyError, ValueError):
        return None
```

from fastapi import Depends, HTTPException, Query
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from auth.utils import decode_token
from db.database import get_db
import aiosqlite

security = HTTPBearer(auto_error=False)

async def get_current_user(
    credentials: HTTPAuthorizationCredentials | None = Depends(security),
    token: str | None = Query(None),
    db: aiosqlite.Connection = Depends(get_db),
):
    # Support both Bearer header and query param (for SSE/EventSource)
    raw_token = None
    if credentials:
        raw_token = credentials.credentials
    elif token:
        raw_token = token

    if not raw_token:
        raise HTTPException(401, "Not authenticated")

    user_id = decode_token(raw_token)
    if user_id is None:
        raise HTTPException(401, "Invalid token")

    cursor = await db.execute("SELECT id, email, language FROM users WHERE id = ?", (user_id,))
    user = await cursor.fetchone()
    if not user:
        raise HTTPException(401, "User not found")
    return dict(user)
```

```python
from fastapi import APIRouter, Depends, HTTPException
from pydantic import BaseModel, EmailStr
from db.database import get_db
from auth.utils import hash_password, verify_password, create_token
import aiosqlite

router = APIRouter(prefix="/api/auth", tags=["auth"])

class AuthRequest(BaseModel):
    email: str
    password: str

class AuthResponse(BaseModel):
    token: str
    user: dict

@router.post("/signup", response_model=AuthResponse)
async def signup(body: AuthRequest, db: aiosqlite.Connection = Depends(get_db)):
    # Check if email exists
    cursor = await db.execute("SELECT id FROM users WHERE email = ?", (body.email,))
    if await cursor.fetchone():
        raise HTTPException(409, "Email already registered")

    hashed = hash_password(body.password)
    cursor = await db.execute(
        "INSERT INTO users (email, password_hash) VALUES (?, ?)",
        (body.email, hashed),
    )
    await db.commit()
    user_id = cursor.lastrowid
    token = create_token(user_id)
    return {"token": token, "user": {"id": user_id, "email": body.email}}

@router.post("/login", response_model=AuthResponse)
async def login(body: AuthRequest, db: aiosqlite.Connection = Depends(get_db)):
    cursor = await db.execute(
        "SELECT id, email, password_hash FROM users WHERE email = ?",
        (body.email,),
    )
    user = await cursor.fetchone()
    if not user or not verify_password(body.password, user["password_hash"]):
        raise HTTPException(401, "Invalid email or password")

    token = create_token(user["id"])
    return {"token": token, "user": {"id": user["id"], "email": user["email"]}}
```

```python
import os
import uuid
from fastapi import APIRouter, Depends, HTTPException, UploadFile, File
from db.database import get_db
from auth.dependencies import get_current_user
from letters.schemas import LetterUploadResponse, LetterSummary, LetterDetail
from config import UPLOAD_DIR, MAX_FILE_SIZE, ALLOWED_CONTENT_TYPES
import aiosqlite

router = APIRouter(prefix="/api/letters", tags=["letters"])

@router.post("/upload", response_model=LetterUploadResponse)
async def upload_letter(
    file: UploadFile = File(...),
    user: dict = Depends(get_current_user),
    db: aiosqlite.Connection = Depends(get_db),
):
    if file.content_type not in ALLOWED_CONTENT_TYPES:
        raise HTTPException(400, f"Unsupported file type: {file.content_type}")

    content = await file.read()
    if len(content) > MAX_FILE_SIZE:
        raise HTTPException(400, "File too large (max 10MB)")

    # Save file
    user_dir = os.path.join(UPLOAD_DIR, str(user["id"]))
    os.makedirs(user_dir, exist_ok=True)
    ext = file.filename.rsplit(".", 1)[-1] if file.filename else "bin"
    filename = f"{uuid.uuid4().hex}.{ext}"
    filepath = os.path.join(user_dir, filename)

    with open(filepath, "wb") as f:
        f.write(content)

    # Insert record
    cursor = await db.execute(
        "INSERT INTO letters (user_id, original_file, status) VALUES (?, ?, 'uploaded')",
        (user["id"], filepath),
    )
    await db.commit()
    return {"letter_id": cursor.lastrowid}

@router.get("", response_model=list[LetterSummary])
async def list_letters(
    user: dict = Depends(get_current_user),
    db: aiosqlite.Connection = Depends(get_db),
):
    cursor = await db.execute(
        "SELECT id, letter_type, risk_score, deadline_date, created_at, status "
        "FROM letters WHERE user_id = ? ORDER BY created_at DESC",
        (user["id"],),
    )
    rows = await cursor.fetchall()
    return [dict(r) for r in rows]

@router.get("/{letter_id}", response_model=LetterDetail)
async def get_letter(
    letter_id: int,
    user: dict = Depends(get_current_user),
    db: aiosqlite.Connection = Depends(get_db),
):
    cursor = await db.execute(
        "SELECT id, letter_type, risk_score, deadline_date, ocr_text, explanation, "
        "response_draft, checklist, citations, consequence, status, created_at "
        "FROM letters WHERE id = ? AND user_id = ?",
        (letter_id, user["id"]),
    )
    row = await cursor.fetchone()
    if not row:
        raise HTTPException(404, "Letter not found")
    return dict(row)
```

        # Step 3 + 4: RAG + Response Generation
        async for event in mock_rag_pipeline(ocr_text, agent_result, language):
            yield sse_event(event.type, event.data)

        yield sse_event("done", {"status": "complete"})

    except Exception as e:

- [ ] **Step 3: Create pipeline router**

Create `backend/pipeline/router.py`:

```python
import json
from fastapi import APIRouter, Depends, HTTPException, Query
from fastapi.responses import StreamingResponse
from db.database import get_db
from auth.dependencies import get_current_user
from pipeline.orchestrator import run_pipeline
import aiosqlite

router = APIRouter(tags=["pipeline"])

@router.get("/api/letters/{letter_id}/process")
async def process_letter(
    letter_id: int,
    token: str = Query(...),
    lang: str = Query("en"),
    user: dict = Depends(get_current_user),
    db: aiosqlite.Connection = Depends(get_db),
):
    # Fetch the letter
    cursor = await db.execute(
        "SELECT id, original_file, status FROM letters WHERE id = ? AND user_id = ?",
        (letter_id, user["id"]),
    )
    letter = await cursor.fetchone()
    if not letter:
        raise HTTPException(404, "Letter not found")

    file_path = letter["original_file"]

    # Update status
    await db.execute(
        "UPDATE letters SET status = 'processing' WHERE id = ?",
        (letter_id,),
    )
    await db.commit()

    async def event_stream():
        collected = {}
        async for event_str in run_pipeline(file_path, lang):
            # Parse event to collect results for DB storage
            if event_str.startswith("event:"):
                lines = event_str.strip().split("\n")
                event_type = lines[0].split(": ", 1)[1]
                data_str = lines[1].split(": ", 1)[1]
                data = json.loads(data_str)
                collected[event_type] = data

            yield event_str

        # Save results to DB
        await db.execute(
            """UPDATE letters SET
                ocr_text = ?,
                letter_type = ?,
                risk_score = ?,
                deadline_date = ?,
                consequence = ?,
                status = 'completed'
            WHERE id = ?""",
            (
                collected.get("ocr_result", {}).get("text"),
                collected.get("classification", {}).get("type"),
                collected.get("risk_score", {}).get("score"),
                collected.get("deadline", {}).get("date"),
                collected.get("consequence", {}).get("text"),
                letter_id,
            ),
        )

        # Create deadline record if deadline found
        deadline_data = collected.get("deadline")
        if deadline_data and deadline_data.get("date"):
            classification = collected.get("classification", {})
            risk = collected.get("risk_score", {})
            await db.execute(
                "INSERT INTO deadlines (letter_id, user_id, title, due_date, risk_score) VALUES (?, ?, ?, ?, ?)",
                (
                    letter_id,
                    user["id"],
                    classification.get("type", "Unknown deadline"),
                    deadline_data["date"],
                    risk.get("score"),
                ),
            )

        await db.commit()

    return StreamingResponse(
        event_stream(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",
            "Connection": "keep-alive",
            "X-Accel-Buffering": "no",
        },
    )
```
