```typescript
const TOKEN_KEY = "klar_token";
const LANG_KEY = "klar_language";

export function getToken(): string | null {
  if (typeof window === "undefined") return null;
  return localStorage.getItem(TOKEN_KEY);
}

export function setToken(token: string): void {
  localStorage.setItem(TOKEN_KEY, token);
}

export function clearToken(): void {
  localStorage.removeItem(TOKEN_KEY);
}

export function isLoggedIn(): boolean {
  return getToken() !== null;
}

export function getLanguage(): string {
  if (typeof window === "undefined") return "en";
  return localStorage.getItem(LANG_KEY) || "en";
}

export function setLanguage(lang: string): void {
  localStorage.setItem(LANG_KEY, lang);
}
```

- [ ] **Step 2: Create API client**

Create `frontend/src/lib/api.ts`:

```typescript
import { API_BASE } from "./constants";
import { getToken } from "./auth";

class ApiError extends Error {
  constructor(public status: number, message: string) {
    super(message);
  }
}

async function request<T>(
  path: string,
  options: RequestInit = {}
): Promise<T> {
  const token = getToken();
  const headers: Record<string, string> = {
    ...(options.headers as Record<string, string>),
  };
  if (token) headers["Authorization"] = `Bearer ${token}`;
  if (!(options.body instanceof FormData)) {
    headers["Content-Type"] = "application/json";
  }

  const res = await fetch(`${API_BASE}${path}`, { ...options, headers });
  if (!res.ok) {
    const body = await res.json().catch(() => ({ detail: res.statusText }));
    throw new ApiError(res.status, body.detail || res.statusText);
  }
  return res.json();
}

export async function signup(email: string, password: string) {
  return request<{ token: string }>("/api/auth/signup", {
    method: "POST",
    body: JSON.stringify({ email, password }),
  });
}

export async function login(email: string, password: string) {
  return request<{ token: string }>("/api/auth/login", {
    method: "POST",
    body: JSON.stringify({ email, password }),
  });
}

export async function uploadLetter(file: File) {
  const form = new FormData();
  form.append("file", file);
  return request<{ letter_id: number }>("/api/letters/upload", {
    method: "POST",
    body: form,
  });
}

export async function getLetters() {
  return request<Array<{
    id: number;
    letter_type: string | null;
    risk_score: number | null;
    deadline_date: string | null;
    created_at: string;
    status: string;
  }>>("/api/letters");
}

export async function getDeadlines() {
  return request<Array<{
    id: number;
    letter_id: number;
    title: string;
    due_date: string;
    status: string;
    risk_score: number;
  }>>("/api/deadlines");
}
```

```tsx
"use client";

import { useState, useRef, useCallback } from "react";
import styles from "./UploadZone.module.css";

interface UploadZoneProps {
  onFileSelected: (file: File) => void;
  disabled?: boolean;
}

const ACCEPTED = ".jpg,.jpeg,.png,.pdf";
const ACCEPTED_TYPES = ["image/jpeg", "image/png", "application/pdf"];

export default function UploadZone({ onFileSelected, disabled }: UploadZoneProps) {
  const [dragOver, setDragOver] = useState(false);
  const [preview, setPreview] = useState<string | null>(null);
  const [fileName, setFileName] = useState<string | null>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);
  const videoRef = useRef<HTMLVideoElement>(null);
  const [cameraActive, setCameraActive] = useState(false);
  const streamRef = useRef<MediaStream | null>(null);

  function handleFile(file: File) {
    if (!ACCEPTED_TYPES.includes(file.type)) {
      alert("Please upload a JPG, PNG, or PDF file.");
      return;
    }
    setFileName(file.name);
    if (file.type.startsWith("image/")) {
      setPreview(URL.createObjectURL(file));
    } else {
      setPreview(null);
    }
    onFileSelected(file);
  }

  function handleDrop(e: React.DragEvent) {
    e.preventDefault();
    setDragOver(false);
    const file = e.dataTransfer.files[0];
    if (file) handleFile(file);
  }

  function handleInputChange(e: React.ChangeEvent<HTMLInputElement>) {
    const file = e.target.files?.[0];
    if (file) handleFile(file);
  }

  const stopCamera = useCallback(() => {
    streamRef.current?.getTracks().forEach((t) => t.stop());
    streamRef.current = null;
    setCameraActive(false);
  }, []);

  async function startCamera() {
    try {
      const stream = await navigator.mediaDevices.getUserMedia({
        video: { facingMode: "environment" },
      });
      streamRef.current = stream;
      setCameraActive(true);
      if (videoRef.current) {
        videoRef.current.srcObject = stream;
        videoRef.current.play();
      }
    } catch {
      alert("Camera access denied or unavailable.");
    }
  }

  function capturePhoto() {
    if (!videoRef.current) return;
    const canvas = document.createElement("canvas");
    canvas.width = videoRef.current.videoWidth;
    canvas.height = videoRef.current.videoHeight;
    canvas.getContext("2d")?.drawImage(videoRef.current, 0, 0);
    canvas.toBlob((blob) => {
      if (blob) {
        const file = new File([blob], "camera-capture.jpg", { type: "image/jpeg" });
        handleFile(file);
        stopCamera();
      }
    }, "image/jpeg", 0.9);
  }

  return (
    <div>
      {cameraActive ? (
        <div style={{ textAlign: "center" }}>
          <video
            ref={videoRef}
            style={{ maxWidth: "100%", borderRadius: "var(--radius)" }}
            autoPlay
            playsInline
          />
          <div className={styles.actions}>
            <button className={`${styles.btn} ${styles.btnPrimary}`} onClick={capturePhoto}>
              Capture
            </button>
            <button className={`${styles.btn} ${styles.btnSecondary}`} onClick={stopCamera}>
              Cancel
            </button>
          </div>
        </div>
      ) : (
        <>
          <div
            className={`${styles.zone} ${dragOver ? styles.zoneDragOver : ""}`}
            onDragOver={(e) => { e.preventDefault(); setDragOver(true); }}
            onDragLeave={() => setDragOver(false)}
            onDrop={handleDrop}
            onClick={() => fileInputRef.current?.click()}
          >
            <input
              ref={fileInputRef}
              type="file"
              accept={ACCEPTED}
              onChange={handleInputChange}
              className={styles.hidden}
            />
            {preview ? (
              <img src={preview} alt="Preview" className={styles.preview} />
            ) : fileName ? (
              <p style={{ fontSize: "1.1rem" }}>{fileName}</p>
            ) : (
              <>
                <p style={{ fontSize: "1.25rem", fontWeight: 600 }}>
                  Drop your letter here
                </p>
                <p style={{ color: "var(--color-text-light)", marginTop: "0.5rem" }}>
                  or click to browse (JPG, PNG, PDF)
                </p>
              </>
            )}
          </div>
          <div className={styles.actions}>
            <button
              className={`${styles.btn} ${styles.btnCamera}`}
              onClick={startCamera}
              disabled={disabled}
            >
              Use Camera
            </button>
          </div>
        </>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useEffect, useState } from "react";
import ReactMarkdown from "react-markdown";
import { API_BASE } from "@/lib/constants";
import { getToken, getLanguage } from "@/lib/auth";
import RiskBadge from "./RiskBadge";
import styles from "./StreamingResults.module.css";

interface StreamingResultsProps {
  letterId: string;
}

interface State {
  ocrText: string;
  classification: { type: string; agency: string } | null;
  riskScore: { score: number; label: string } | null;
  deadline: { date: string; days_remaining: number } | null;
  consequence: string;
  explanation: string;
  responseDraft: string;
  checklist: string[];
  citations: { section: string; text: string }[];
  done: boolean;
  error: string;
}

const INITIAL: State = {
  ocrText: "",
  classification: null,
  riskScore: null,
  deadline: null,
  consequence: "",
  explanation: "",
  responseDraft: "",
  checklist: [],
  citations: [],
  done: false,
  error: "",
};

export default function StreamingResults({ letterId }: StreamingResultsProps) {
  const [state, setState] = useState<State>(INITIAL);
  const [ocrOpen, setOcrOpen] = useState(false);
  const [citationsOpen, setCitationsOpen] = useState(false);
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    const token = getToken();
    const lang = getLanguage();
    const url = `${API_BASE}/api/letters/${letterId}/process?token=${token}&lang=${lang}`;
    const es = new EventSource(url);

    es.addEventListener("ocr_result", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, ocrText: data.text }));
    });
    es.addEventListener("classification", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, classification: data }));
    });
    es.addEventListener("risk_score", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, riskScore: data }));
    });
    es.addEventListener("deadline", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, deadline: data }));
    });
    es.addEventListener("consequence", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, consequence: data.text }));
    });
    es.addEventListener("explanation", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, explanation: s.explanation + data.chunk }));
    });
    es.addEventListener("response_draft", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, responseDraft: s.responseDraft + data.chunk }));
    });
    es.addEventListener("checklist", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, checklist: data.items }));
    });
    es.addEventListener("citations", (e) => {
      const data = JSON.parse(e.data);
      setState((s) => ({ ...s, citations: data.items }));
    });
    es.addEventListener("done", () => {
      setState((s) => ({ ...s, done: true }));
      es.close();
    });
    es.addEventListener("error", (e) => {
      if (e instanceof MessageEvent) {
        const data = JSON.parse(e.data);
        setState((s) => ({ ...s, error: data.message }));
      }
      es.close();
    });

    return () => es.close();
  }, [letterId]);

  function copyResponse() {
    navigator.clipboard.writeText(state.responseDraft);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  }

  const loading = !state.done && !state.error;

  return (
    <div>
      {/* Classification + Risk */}
      {state.classification && (
        <div className={styles.section} style={{ display: "flex", alignItems: "center", gap: "1rem", flexWrap: "wrap" }}>
          <span style={{ fontSize: "1.1rem", fontWeight: 600 }}>
            {state.classification.type}
          </span>
          {state.riskScore && <RiskBadge score={state.riskScore.score} />}
        </div>
      )}

      {/* Deadline */}
      {state.deadline && (
        <div className={styles.section}>
          <div className={styles.sectionHeader}>Deadline</div>
          <p style={{ fontSize: "1.1rem", fontWeight: 600 }}>
            {state.deadline.date}
            <span style={{
              marginLeft: "0.75rem",
              color: state.deadline.days_remaining <= 7 ? "var(--color-danger)" : "var(--color-text-light)",
            }}>
              ({state.deadline.days_remaining} days remaining)
            </span>
          </p>
        </div>
      )}

      {/* Consequence */}
      {state.consequence && (
        <div className={`${styles.section} ${styles.consequence}`}>
          <div className={styles.sectionHeader}>What happens if you miss it</div>
          <p>{state.consequence}</p>
        </div>
      )}

      {/* Explanation */}
      {(state.explanation || loading) && (
        <div className={styles.section}>
          <div className={styles.sectionHeader}>
            Explanation {loading && !state.explanation && <span className={styles.spinner} />}
          </div>
          {state.explanation && <ReactMarkdown>{state.explanation}</ReactMarkdown>}
        </div>
      )}

      {/* Response Draft */}
      {state.responseDraft && (
        <div className={`${styles.section} ${styles.responseDraft}`}>
          <div className={styles.sectionHeader}>Your Response (ready to send)</div>
          <pre style={{ whiteSpace: "pre-wrap", fontFamily: "inherit", lineHeight: 1.6 }}>
            {state.responseDraft}
          </pre>
          <button className={styles.copyBtn} onClick={copyResponse}>
            {copied ? "Copied!" : "Copy to clipboard"}
          </button>
        </div>
      )}

      {/* Checklist */}
      {state.checklist.length > 0 && (
        <div className={styles.section}>
          <div className={styles.sectionHeader}>Document Checklist</div>
          {state.checklist.map((item, i) => (
            <label key={i} className={styles.checklistItem}>
              <input type="checkbox" />
              <span>{item}</span>
            </label>
          ))}
        </div>
      )}

      {/* Citations */}
      {state.citations.length > 0 && (
        <div className={styles.section}>
          <div
            className={`${styles.sectionHeader} ${styles.collapsible}`}
            onClick={() => setCitationsOpen(!citationsOpen)}
          >
            Legal References ({state.citations.length}) {citationsOpen ? "▼" : "▶"}
          </div>
          <div className={`${styles.collapsibleContent} ${citationsOpen ? styles.collapsibleContentOpen : ""}`}>
            {state.citations.map((c, i) => (
              <div key={i} style={{ padding: "0.5rem 0", borderBottom: "1px solid var(--color-border)" }}>
                <strong>{c.section}</strong>
                <p style={{ color: "var(--color-text-light)", fontSize: "0.9rem" }}>{c.text}</p>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* OCR (collapsible) */}
      {state.ocrText && (
        <div className={styles.section}>
          <div
            className={`${styles.sectionHeader} ${styles.collapsible}`}
            onClick={() => setOcrOpen(!ocrOpen)}
          >
            Original Text (OCR) {ocrOpen ? "▼" : "▶"}
          </div>
          <div className={`${styles.collapsibleContent} ${ocrOpen ? styles.collapsibleContentOpen : ""}`}>
            <pre style={{ whiteSpace: "pre-wrap", fontFamily: "inherit", fontSize: "0.9rem", color: "var(--color-text-light)" }}>
              {state.ocrText}
            </pre>
          </div>
        </div>
      )}

      {/* Loading indicator */}
      {loading && !state.classification && (
        <div style={{ textAlign: "center", padding: "3rem" }}>
          <span className={styles.spinner} style={{ width: 32, height: 32 }} />
          <p style={{ marginTop: "1rem", color: "var(--color-text-light)" }}>
            Processing your letter...
          </p>
        </div>
      )}

      {/* Error */}
      {state.error && (
        <div className={`${styles.section} ${styles.consequence}`}>
          <div className={styles.sectionHeader}>Error</div>
          <p>{state.error}</p>
        </div>
      )}

      {/* Done */}
      {state.done && (
        <p style={{ textAlign: "center", color: "var(--color-success)", fontWeight: 600, marginTop: "1rem" }}>
          Processing complete
        </p>
      )}
    </div>
  );
}
```

```tsx
import Link from "next/link";
import RiskBadge from "./RiskBadge";
import styles from "./DeadlineCard.module.css";

interface DeadlineCardProps {
  letterId: number;
  letterType: string | null;
  deadlineDate: string | null;
  riskScore: number | null;
  status: string;
  createdAt: string;
}

export default function DeadlineCard({
  letterId,
  letterType,
  deadlineDate,
  riskScore,
  status,
  createdAt,
}: DeadlineCardProps) {
  const daysLeft = deadlineDate
    ? Math.ceil((new Date(deadlineDate).getTime() - Date.now()) / 86400000)
    : null;

  const isOverdue = daysLeft !== null && daysLeft < 0;
  const isUrgent = daysLeft !== null && daysLeft >= 0 && daysLeft <= 7;

  const cardClass = [
    styles.card,
    isOverdue ? styles.overdue : "",
    isUrgent ? styles.urgent : "",
  ].join(" ");

  return (
    <Link href={`/results/${letterId}`} style={{ textDecoration: "none", color: "inherit" }}>
      <div className={cardClass}>
        <div className={styles.left}>
          <div className={styles.type}>
            {letterType || "Processing..."}
          </div>
          <div className={styles.meta}>
            {deadlineDate ? `Deadline: ${deadlineDate}` : "No deadline"} &middot;{" "}
            Status: {status} &middot; Uploaded: {new Date(createdAt).toLocaleDateString()}
          </div>
        </div>
        <div className={styles.right}>
          {riskScore && <RiskBadge score={riskScore} />}
          {daysLeft !== null && (
            <div
              className={styles.daysLeft}
              style={{ color: isOverdue ? "var(--color-danger)" : isUrgent ? "var(--color-accent)" : "var(--color-text)" }}
            >
              {isOverdue ? `${Math.abs(daysLeft)}d overdue` : `${daysLeft}d left`}
            </div>
          )}
        </div>
      </div>
    </Link>
  );
}
```

```tsx
  risk_score: number | null;
  deadline_date: string | null;
  created_at: string;
  status: string;
}

export default function DashboardPage() {
  const [letters, setLetters] = useState<Letter[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState("");

  useEffect(() => {
    getLetters()
      .then(setLetters)
      .catch((err) => setError(err.message))
      .finally(() => setLoading(false));
  }, []);

  const sorted = [...letters].sort((a, b) => {
    if (!a.deadline_date) return 1;
    if (!b.deadline_date) return -1;
    return new Date(a.deadline_date).getTime() - new Date(b.deadline_date).getTime();
  });

  return (
    <main className="container container--wide" style={{ marginTop: "2rem", paddingBottom: "3rem" }}>
      <div style={{ display: "flex", justifyContent: "space-between", alignItems: "center", marginBottom: "1.5rem" }}>
        <h1 style={{ color: "var(--color-primary)" }}>Your Letters</h1>
        <Link
          href="/upload"
          style={{
            background: "var(--color-primary)",
            color: "#fff",
            padding: "0.5rem 1.25rem",
            borderRadius: "var(--radius)",
            fontWeight: 600,
          }}
        >
          Upload new
        </Link>
      </div>

      {loading && <p>Loading...</p>}
      {error && <p style={{ color: "var(--color-danger)" }}>{error}</p>}

      {!loading && letters.length === 0 && (
        <div style={{ textAlign: "center", padding: "4rem 2rem" }}>
          <p style={{ fontSize: "1.1rem", color: "var(--color-text-light)" }}>
            No letters yet. Upload your first German letter to get started.
          </p>
        </div>
      )}

      <div style={{ display: "flex", flexDirection: "column", gap: "0.75rem" }}>
        {sorted.map((letter) => (
          <DeadlineCard
            key={letter.id}
            letterId={letter.id}
            letterType={letter.letter_type}
            deadlineDate={letter.deadline_date}
            riskScore={letter.risk_score}
            status={letter.status}
            createdAt={letter.created_at}
          />
        ))}
      </div>
    </main>
  );
}
```
